cmake_minimum_required(VERSION 3.16)
project(PasswordValidator VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
add_library(validator src/password_validator.cpp)
target_include_directories(validator PUBLIC src)
target_link_libraries(validator PUBLIC Qt6::Core)
add_executable(password_gui src/main.cpp)
target_link_libraries(password_gui PRIVATE validator Qt6::Widgets)
add_executable(password_demo src/demo.cpp)
target_link_libraries(password_demo PRIVATE validator)
add_executable(password_tests tests/tests.cpp)
target_link_libraries(password_tests PRIVATE validator)
enable_testing()
add_test(NAME validator_tests COMMAND password_tests)
if(APPLE)
    set_target_properties(password_gui PROPERTIES MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_GUI_IDENTIFIER "edu.password.validator"
        MACOSX_BUNDLE_BUNDLE_NAME "Валидатор пароля")
endif()
